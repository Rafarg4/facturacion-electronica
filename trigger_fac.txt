DELIMITER $$

CREATE TRIGGER trg_productos_ai
AFTER INSERT ON productos
FOR EACH ROW
BEGIN

INSERT INTO auditoria(
    tabla,
    registro_id,
    accion,
    usuario_mysql,
    datos_nuevos
)
VALUES(
    'productos',
    NEW.id,
    'INSERT',
    CURRENT_USER(),

    JSON_OBJECT(
        'id', NEW.id,
        'codigo', NEW.codigo,
        'num_item', NEW.num_item,
        'descripcion', NEW.descripcion,
        'imagen', NEW.imagen,
        'id_rubro', NEW.id_rubro,
        'cantidad', NEW.cantidad,
        'costo', NEW.costo,
        'precio1', NEW.precio1,
        'precio2', NEW.precio2,
        'precio3', NEW.precio3,
        'cantidad_minima', NEW.cantidad_minima,
        'cantidad_caja', NEW.cantidad_caja,
        'estado', NEW.estado,
        'tipo_iva', NEW.tipo_iva,
        'tipo_moneda', NEW.tipo_moneda
    )
);

END$$
